通用实体

    @TableId(value = "id", type = IdType.AUTO)
    private Integer id;

    /**
     * 客户端版本号
     */
    @TableField("clientVersion")
    @Schema(description = "客户端版本号", example = "1")
    private String clientVersion;

    /**
     * 开关（0关闭 ,1 开启）
     */
    @TableField("switchType")
    @Schema(description = "开关（0关闭 ,1 开启）", example = "1")
    private Integer switchType;

    /**
     * 创建人
     */
    @TableField("createUser")
    @Schema(description = "创建人", example = "text")
    private String createUser;

    /**
     * 创建时间
     */
    @TableField("createTime")
    @Schema(description = "创建时间", example = "2018-01-24 14:55:11")
    private String createTime;

    /**
     * 修改时间
     */
    @TableField("updateTime")
    @Schema(description = "修改时间", example = "2018-01-24 14:55:11")
    private String updateTime;

    /**
     * 状态 1.正常 0.删除
     */
    @TableField("status")
    @Schema(description = "状态 1.正常 0.删除", example = "1")
    private Integer status;
