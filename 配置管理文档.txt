    @TableId(value = "ID", type = IdType.AUTO)
    private Integer id;

    /**
     * 配置项名称 
     */
    @Schema(description = "配置项名称", example = "名称")
    @TableField("ConfigName")
    private String ConfigName;

    /**
     * 配置项识别码 
     */
    @Schema(description = "配置项识别码", example = "识别码")
    @TableField("ConfigCode")
    private String ConfigCode;

    /**
     * 配置项内容
     */
    @Schema(description = "配置项内容", example = "内容")
    @TableField("ConfigInfo")
    private String ConfigInfo;

    /**
     * 配置项说明
     */
    @Schema(description = "配置项说明", example = "说明")
    @TableField("ConfigNote")
    private String ConfigNote;

    /**
     * 状态 1.有效、0.无效
     */
    @Schema(description = "状态 1.有效、0.无效", example = "1")
    @TableField("Status")
    private Integer Status;

    /**
     * 创建时间
     */
    @Schema(description = "创建时间", example = "2023-01-01 00:00:00")
    @TableField("CreateTime")
    @JsonFormat(pattern = "yyyy-MM-dd HH:mm:ss", timezone = "GMT+8")
    private LocalDateTime CreateTime;

    /**
     * 更新时间
     */
    @Schema(description = "更新时间", example = "2023-01-01 00:00:00")
    @TableField("UpdateTime")
    @JsonFormat(pattern = "yyyy-MM-dd HH:mm:ss", timezone = "GMT+8")
    private LocalDateTime UpdateTime;

    /**
     * 用户名
     */
    @Schema(description = "用户名", example = "用户名")
    @TableField("UserName")
    private String UserName;

    /**
     * 客户端版本
     */
    @Schema(description = "客户端版本", example = "客户端版本")
    @TableField("ConfigClientVersion")
    private String ConfigClientVersion;

    /**
     * 配置项筛选条件
     */
    @Schema(description = "配置项筛选条件", example = "筛选条件")
    @TableField("ConfigCondition")
    private String ConfigCondition;

    /**
     * 配置项内容格式
     */
    @Schema(description = "配置项内容格式", example = "内容格式")
    @TableField("ConfigInfoFormat")
    private String ConfigInfoFormat;
